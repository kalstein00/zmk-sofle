if(CONFIG_SHIELD_SOFLE_DONGLE)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(dongle_original "${APPLICATION_SOURCE_DIR}/src/split/bluetooth/central.c")
  set(dongle_generated "${CMAKE_BINARY_DIR}/dongle_split_central.c")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${dongle_original}" "${CMAKE_CURRENT_LIST_DIR}/prepare_dongle_central.py")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/prepare_dongle_central.py" "${dongle_original}" "${dongle_generated}"
    RESULT_VARIABLE dongle_hook_result)
  if(NOT dongle_hook_result EQUAL 0)
    message(FATAL_ERROR "Unable to prepare pinned dongle split central")
  endif()
  set(dongle_display_original "${APPLICATION_SOURCE_DIR}/src/display/main.c")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${dongle_display_original}" "${CMAKE_CURRENT_LIST_DIR}/prepare_dongle_display.py")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/prepare_dongle_display.py" "${dongle_display_original}" "${CMAKE_BINARY_DIR}/dongle_display_main.c" RESULT_VARIABLE display_hook_result)
  if(NOT display_hook_result EQUAL 0)
    message(FATAL_ERROR "Unable to prepare pinned dongle display")
  endif()
  function(dongle_replace_central_source)
    get_target_property(dongle_sources app SOURCES)
    set(dongle_replaced 0)
    set(display_replaced 0)
    set(dongle_new_sources)
    foreach(dongle_source IN LISTS dongle_sources)
      if(dongle_source MATCHES "(^|/)src/split/bluetooth/central\\.c(>|$)")
        list(APPEND dongle_new_sources "${CMAKE_BINARY_DIR}/dongle_split_central.c")
        math(EXPR dongle_replaced "${dongle_replaced}+1")
      elseif(dongle_source MATCHES "(^|/)src/display/main\\.c(>|$)")
        list(APPEND dongle_new_sources "${CMAKE_BINARY_DIR}/dongle_display_main.c")
        math(EXPR display_replaced "${display_replaced}+1")
      else()
        list(APPEND dongle_new_sources "${dongle_source}")
      endif()
    endforeach()
    if(NOT dongle_replaced EQUAL 1)
      message(FATAL_ERROR "Expected exactly one split Bluetooth central source; found ${dongle_replaced}")
    endif()
    if(NOT display_replaced EQUAL 1)
      message(FATAL_ERROR "Expected exactly one display source; found ${display_replaced}")
    endif()
    set_property(TARGET app PROPERTY SOURCES "${dongle_new_sources}")
  endfunction()
  cmake_language(DEFER DIRECTORY "${APPLICATION_SOURCE_DIR}" CALL dongle_replace_central_source)
  target_include_directories(app PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${APPLICATION_SOURCE_DIR}/src/display")
  zephyr_library()
  zephyr_library_include_directories(${APPLICATION_SOURCE_DIR}/include)
  zephyr_library_sources(dongle_usb_only.c dongle_slot_state.c dongle_battery_snapshot.c dongle_display.c dongle_sleep.c bongo_cat_images.c)
  zephyr_ld_options(-Wl,--wrap=bt_le_adv_start)
endif()
