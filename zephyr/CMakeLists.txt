if(CONFIG_SHIELD_SOFLE_DONGLE)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(wireless_names ble battery endpoints hog usb_hid behavior_queue split_central sticky_key two_axis studio_rpc)
  set(wireless_rel_studio_rpc "studio/rpc")
  set(wireless_rel_split_central "split/central")
  set(wireless_rel_sticky_key "behaviors/behavior_sticky_key")
  set(wireless_rel_two_axis "behaviors/behavior_input_two_axis")
  foreach(wireless_name IN LISTS wireless_names)
    if(NOT DEFINED wireless_rel_${wireless_name})
      set(wireless_rel_${wireless_name} "${wireless_name}")
    endif()
  endforeach()
  if(CONFIG_SHIELD_SOFLE_DONGLE_WIRELESS)
    target_include_directories(app PRIVATE "${APPLICATION_SOURCE_DIR}/src/studio")
    foreach(wireless_name IN LISTS wireless_names)
      set(wireless_input "${APPLICATION_SOURCE_DIR}/src/${wireless_rel_${wireless_name}}.c")
      set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${wireless_input}" "${CMAKE_CURRENT_LIST_DIR}/prepare_dongle_wireless.py")
      execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/prepare_dongle_wireless.py"
        "${wireless_name}" "${wireless_input}" "${CMAKE_BINARY_DIR}/dongle_wireless_${wireless_name}.c"
        RESULT_VARIABLE wireless_result)
      if(NOT wireless_result EQUAL 0)
        message(FATAL_ERROR "Unable to prepare pinned wireless ${wireless_name}")
      endif()
      set_property(TARGET app APPEND PROPERTY SOFLE_WIRELESS_ADAPTERS "${wireless_name}")
      set_property(TARGET app PROPERTY SOFLE_WIRELESS_SOURCE_${wireless_name} "${wireless_rel_${wireless_name}}")
    endforeach()
  endif()
  set(dongle_original "${APPLICATION_SOURCE_DIR}/src/split/bluetooth/central.c")
  set(dongle_generated "${CMAKE_BINARY_DIR}/dongle_split_central.c")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${dongle_original}" "${CMAKE_CURRENT_LIST_DIR}/prepare_dongle_central.py")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/prepare_dongle_central.py" "${dongle_original}" "${dongle_generated}"
    RESULT_VARIABLE dongle_hook_result)
  if(NOT dongle_hook_result EQUAL 0)
    message(FATAL_ERROR "Unable to prepare pinned dongle split central")
  endif()
  set(dongle_display_original "${APPLICATION_SOURCE_DIR}/src/display/main.c")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${dongle_display_original}" "${CMAKE_CURRENT_LIST_DIR}/prepare_dongle_display.py")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/prepare_dongle_display.py" "${dongle_display_original}" "${CMAKE_BINARY_DIR}/dongle_display_main.c" RESULT_VARIABLE display_hook_result)
  if(NOT display_hook_result EQUAL 0)
    message(FATAL_ERROR "Unable to prepare pinned dongle display")
  endif()
  function(dongle_replace_central_source)
    get_target_property(dongle_sources app SOURCES)
    if(CONFIG_SHIELD_SOFLE_DONGLE_WIRELESS)
      get_target_property(wireless_names app SOFLE_WIRELESS_ADAPTERS)
      if(NOT wireless_names)
        message(FATAL_ERROR "Missing generated wireless source list")
      endif()
      foreach(wireless_name IN LISTS wireless_names)
        get_target_property(wireless_rel_${wireless_name} app SOFLE_WIRELESS_SOURCE_${wireless_name})
        set(wireless_replaced_${wireless_name} 0)
      endforeach()
    endif()
    set(dongle_replaced 0)
    set(display_replaced 0)
    set(dongle_new_sources)
    foreach(dongle_source IN LISTS dongle_sources)
      if(dongle_source MATCHES "(^|/)src/split/bluetooth/central\\.c(>|$)")
        list(APPEND dongle_new_sources "${CMAKE_BINARY_DIR}/dongle_split_central.c")
        math(EXPR dongle_replaced "${dongle_replaced}+1")
      elseif(dongle_source MATCHES "(^|/)src/display/main\\.c(>|$)")
        list(APPEND dongle_new_sources "${CMAKE_BINARY_DIR}/dongle_display_main.c")
        math(EXPR display_replaced "${display_replaced}+1")
      else()
        if(CONFIG_SHIELD_SOFLE_DONGLE_WIRELESS)
          foreach(wireless_name IN LISTS wireless_names)
            if(dongle_source MATCHES "(^|/)src/${wireless_rel_${wireless_name}}\\.c(>|$)")
              set(dongle_source "${CMAKE_BINARY_DIR}/dongle_wireless_${wireless_name}.c")
              math(EXPR wireless_replaced_${wireless_name} "${wireless_replaced_${wireless_name}}+1")
            endif()
          endforeach()
        endif()
        list(APPEND dongle_new_sources "${dongle_source}")
      endif()
    endforeach()
    if(NOT dongle_replaced EQUAL 1)
      message(FATAL_ERROR "Expected exactly one split Bluetooth central source; found ${dongle_replaced}")
    endif()
    if(NOT display_replaced EQUAL 1)
      message(FATAL_ERROR "Expected exactly one display source; found ${display_replaced}")
    endif()
    if(CONFIG_SHIELD_SOFLE_DONGLE_WIRELESS)
      foreach(wireless_name IN LISTS wireless_names)
        if(NOT wireless_replaced_${wireless_name} EQUAL 1)
          message(FATAL_ERROR "Expected one wireless source replacement for ${wireless_name}; found ${wireless_replaced_${wireless_name}}")
        endif()
      endforeach()
    endif()
    set_property(TARGET app PROPERTY SOURCES "${dongle_new_sources}")
  endfunction()
  cmake_language(DEFER DIRECTORY "${APPLICATION_SOURCE_DIR}" CALL dongle_replace_central_source)
  target_include_directories(app PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${APPLICATION_SOURCE_DIR}/src/display")
  zephyr_library()
  zephyr_library_include_directories(${APPLICATION_SOURCE_DIR}/include)
  zephyr_library_sources(dongle_slot_state.c dongle_battery_snapshot.c dongle_display.c dongle_sleep.c bongo_cat_images.c)
  if(CONFIG_SHIELD_SOFLE_DONGLE_WIRELESS)
    zephyr_library_sources(dongle_host.c dongle_host_state.c dongle_menu.c dongle_log.c behavior_host_clear.c behavior_host_menu.c)
  else()
    zephyr_library_sources(dongle_usb_only.c)
    zephyr_ld_options(-Wl,--wrap=bt_le_adv_start)
  endif()
endif()
